### A.1 load.cjs —— 只读 TS 加载器

~~~javascript
const fs = require('fs');
const path = require('path');
const ts = require(path.join(process.cwd(), 'node_modules', 'typescript'));
const cache = new Map();
function loadFile(p) {
  if (cache.has(p)) return cache.get(p).exports;
  const src = fs.readFileSync(p, 'utf8');
  const out = ts.transpileModule(src, {
    compilerOptions: { module: ts.ModuleKind.CommonJS, target: ts.ScriptTarget.ES2022, esModuleInterop: true },
  }).outputText;
  const mod = { exports: {} };
  cache.set(p, mod);
  const dir = path.dirname(p);
  const req = (spec) => {
    if (!spec.startsWith('.')) return require(spec);
    const base = path.resolve(dir, spec);
    const cand = base.endsWith('.ts') ? base : base + '.ts';
    if (fs.existsSync(cand)) return loadFile(cand);
    const idx = path.join(base, 'index.ts');
    if (fs.existsSync(idx)) return loadFile(idx);
    throw new Error('cannot resolve ' + spec + ' from ' + p);
  };
  new Function('exports', 'require', 'module', '__filename', '__dirname', out)(
    mod.exports, req, mod, p, dir);
  return mod.exports;
}
module.exports = { loadFile };
~~~

  if (!c.no || !/\d/.test(c.no)) {
    const i = hay.indexOf('「' + c.name + '」');

    const s = Math.max(0, hay.lastIndexOf('【', i));
    return { stage: null, excerpt: hay.slice(s, s + 80).replace(/\s+/g, ' ') };

    const st = win.match(/Stage\s*([0-9]{1,2})/);
    if (st) return { stage: Number(st[1]), excerpt: win.replace(/\s+/g, ' ').slice(0, 80) };
